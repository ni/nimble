## Button Guidelines

<NimbleThemeProvider theme={theme}>
	<StoryGuidanceGrid>
		<Do
			title="Create a clear action hierarchy"
			example={
				<>
					<NimbleButton appearance="ghost">Cancel</NimbleButton>
					<NimbleButton appearance="block">Save</NimbleButton>
				</>
			}
		>
			Make Save more prominent than Cancel so the intended next
			step is easy to find.
		</Do>
		<Dont
			title="Emphasize multiple competing actions"
			example={
				<>
					<NimbleButton appearance="block">Save</NimbleButton>
					<NimbleButton appearance="block">Submit</NimbleButton>
				</>
			}
		>
			Save and Submit have equal emphasis here, leaving the intended
			next step unclear.
		</Dont>
		<Do
			title="Use concise labels"
			example={<NimbleButton appearance="outline">Save Changes</NimbleButton>}
		>
			Name the action and, when needed, its object: Save Changes.
		</Do>
		<Dont
			title="Use vague labels"
			example={<NimbleButton appearance="outline">Click Here</NimbleButton>}
		>
			Click Here describes the interaction, not its result.
		</Dont>
	</StoryGuidanceGrid>
</NimbleThemeProvider>
