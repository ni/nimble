	<StorySection id="button-guidelines" title="Button Guidelines">
		<StoryGuidanceGrid>
			<Do
				title="Create a clear action hierarchy"
				example={
					<>
						<NimbleButton appearance="ghost">Cancel</NimbleButton>
						<NimbleButton appearance="block">Save</NimbleButton>
					</>
				}
			>
				Use one visually prominent action per section and pair it with
				less prominent actions when users need a clear hierarchy.
			</Do>
			<Dont
				title="Prioritize one prominent action"
				example={
					<>
						<NimbleButton appearance="block">Save</NimbleButton>
						<NimbleButton appearance="block">Submit</NimbleButton>
					</>
				}
			>
				Do not use multiple visually prominent buttons together when
				one action should take priority.
			</Dont>
			<Do
				title="Use concise labels"
				example={
					<NimbleButton appearance="outline">Save Changes</NimbleButton>
				}
			>
				Use concise labels that describe the action the button performs.
			</Do>
			<Dont
				title="Avoid vague labels"
				example={
					<NimbleButton appearance="outline">Click Here</NimbleButton>
				}
			>

				the button is activated.
